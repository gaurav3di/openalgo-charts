### A descriptor with several outputs

The 105 built-ins are not special-cased. Each is a descriptor - inputs, plots and a
`calc` - and anything you register with `registerIndicator` uses the same contract,
so it gets generated settings, a legend entry and state persistence for free.

This one computes a trend regime, then uses four descriptor hooks on the result:
`colorBy` for the line, `barColors` to repaint the candles, `background` to shade the
pane, and `markers` to flag each flip. No fork, no build step.

<RunnableExample height={340} tiers={['indicators']} code={`lib.registerIndicator({
  id: 'demo-regime',
  name: 'Trend Regime',
  category: 'Custom',
  placement: 'onchart',
  inputs: [
    { key: 'length', type: 'number', label: 'Length', default: 20, min: 2, max: 200, step: 1 },
    { key: 'up', type: 'color', label: 'Bull', default: '#26a69a' },
    { key: 'down', type: 'color', label: 'Bear', default: '#ef5350' },
  ],
  plots: [{
    key: 'basis', type: 'line', title: 'Basis', style: { lineWidth: 2 },
    // colorBy is a function, not the name of a column. It runs per bar and
    // returns a colour, or undefined to fall back to the plot's own.
    colorBy: ({ index, values, settings }) => {
      const t = values.trend[index];
      return t == null ? undefined : t > 0 ? settings.up : settings.down;
    },
  }],
  calc(bars, s) {
    const close = lib.sourceValues(bars, 'close');
    const basis = lib.nulls(lib.sma(close, Math.max(2, s.length | 0)));
    const trend = basis.map((b, i) => (b == null ? null : close[i] >= b ? 1 : -1));
    return { basis, trend };
  },
  // Repaint the price candles by regime.
  barColors({ values, settings }) {
    return values.trend.map((t) => (t == null ? null : t > 0 ? settings.up : settings.down));
  },
  // Shade the whole pane behind them.
  background({ values, settings }) {
    return values.trend.map((t) =>
      t == null ? null : lib.withAlpha(t > 0 ? settings.up : settings.down, 0.07));
  },
  // Flag every flip, anchored to the candle rather than to the indicator line.
  markers({ bars, values, settings }) {
    const out = [];
    for (let i = 1; i < bars.length; i++) {
      const a = values.trend[i - 1], b = values.trend[i];
      if (a == null || b == null || a === b) continue;
      out.push({
        time: bars[i].time,
        position: 'atPrice',
        price: b > 0 ? bars[i].low : bars[i].high,
        shape: b > 0 ? 'arrowUp' : 'arrowDown',
        color: b > 0 ? settings.up : settings.down,
        text: b > 0 ? 'BULL' : 'BEAR',
      });
    }
    return out;
  },
});

const chart = lib.createChart(el);
const bars = lib.generateBars(1700000000, 200, 3600);
chart.addSeries('candlestick').setData(bars);
chart.addIndicator('demo-regime');
chart.timeScale.fitContent(bars.length);
return chart;`} />

A descriptor can also supply `draws` for free-standing lines, boxes and labels,
`levels` for data-derived horizontals, `fills` between plots, `table` for an on-chart
grid, and `alerts` the chart evaluates for you. See

### A higher timeframe from the chart's own bars

`securitySeries` folds the bars in view up to a coarser interval, one value per bar, so a
study can read the day's range on an hourly chart without a second data source. The default
reading is the bucket as it stood at each bar, so it never repaints. This one shades the
day's high to low, and pins a `cross` marker to the top edge of the pane on each new day.

<RunnableExample height={340} tiers={['indicators']} code={`lib.registerIndicator({
  id: 'demo-daily-range',
  name: 'Daily Range',
  category: 'Custom',
  placement: 'onchart',
  inputs: [],
  plots: [
    { key: 'high', type: 'step', title: 'Day high', style: { color: '#26a69a', lineWidth: 1 } },
    { key: 'low',  type: 'step', title: 'Day low',  style: { color: '#ef5350', lineWidth: 1 } },
  ],
  fills: [{ between: ['high', 'low'], colorUp: '#4f8cff', colorDown: '#4f8cff', opacity: 0.08 }],
  calc(bars, s) {
    // The chart's zone rides along under the reserved timezone key.
    const day = lib.securitySeries(bars, '1d', { timezone: s.timezone });
    // isNew is a column too: markers reads it below.
    return { high: day.high, low: day.low, isNew: day.isNew.map((v) => (v ? 1 : null)) };
  },
  markers({ bars, values }) {
    const out = [];
    for (let i = 0; i < bars.length; i++) {
      if (values.isNew[i] === null) continue;
      out.push({ time: bars[i].time, position: 'paneTop', shape: 'cross', size: 'small', color: '#8892a6' });
    }
    return out;
  },
});

const chart = lib.createChart(el);
const bars = lib.generateBars(1700000000, 240, 3600);
chart.addSeries('candlestick').setData(bars);
chart.addIndicator('demo-daily-range');
chart.timeScale.fitContent(bars.length);
return chart;`} />

### Calculate an hourly average on minute bars

The average is calculated on hourly bars before alignment to five-minute bars.
The orange line holds the last completed hour's average. The blue line updates
with the developing hour. Both use the same supplied data and a three-hour window.

<RunnableExample height={340} tiers={['indicators']} code={`lib.registerIndicator({
  id: 'demo-requested-mean', name: 'Hourly mean', placement: 'onchart', inputs: [],
  plots: [
    { key: 'confirmed', type: 'step', title: 'Confirmed hour', style: { color: '#e8a23a', lineWidth: 2 } },
    { key: 'developing', type: 'line', title: 'Developing hour', style: { color: '#4f8cff', lineWidth: 2 } },
  ],
  calc(bars, settings) {
    const expression = requested => ({ mean: lib.nulls(lib.sma(requested.map(bar => bar.close), 3)) });
    const options = { timezone: settings.timezone };
    return {
      confirmed: lib.securityExpression(bars, '1h', expression, options).mean ?? [],
      developing: lib.securityExpression(bars, '1h', expression, { ...options, mode: 'developing' }).mean ?? [],
    };
  },
});
const chart = lib.createChart(el, { timezone: 'Etc/UTC' });
const bars = lib.generateBars(1700002800, 240, 300);
chart.addSeries('candlestick').setData(bars);
chart.addIndicator('demo-requested-mean');
chart.fitContent();
return chart;`} />

## Computed fills and multiple summary grids

A descriptor can color each part of a band and publish several tables. The fill
callback reads calculation indices before any plot offset. A per-bar color takes
precedence over the gradient; returning `undefined` uses the gradient. Named grids
keep their identity across updates and disappear when omitted from `tables`.

<RunnableExample height={340} code={`lib.registerIndicator({
  id: 'demo-visual-outputs', name: 'Range summary', placement: 'onchart', inputs: [],
  plots: [
    { key: 'upper', type: 'line', title: 'Upper', style: { color: '#4f8cff' } },
    { key: 'lower', type: 'line', title: 'Lower', style: { color: '#4f8cff' } },
  ],
  calc: bars => ({ upper: bars.map(b => b.high + 3), lower: bars.map(b => b.low - 3) }),
  fills: [{
    between: ['upper', 'lower'], opacity: 0.2,
    gradient: { topColor: '#26a69a', bottomColor: '#4f8cff' },
    colorBy: ({ index }) => index % 20 < 5 ? '#e8a23a' : undefined,
  }],
  tables: ({ bars }) => bars.length ? [
    { id: 'quote', options: { position: 'bottom-center', cellWidth: 120 },
      rows: [[{ text: 'Latest close' }], [{ text: bars[bars.length - 1].close.toFixed(2) }]] },
    { id: 'count', options: { position: 'bottom-right', cellWidth: 120 },
      rows: [[{ text: 'Observations' }], [{ text: String(bars.length) }]] },
  ] : [],
});
const chart = lib.createChart(el);
const bars = lib.generateBars(1700000000, 120, 3600);
chart.addSeries('candlestick').setData(bars);
chart.addIndicator('demo-visual-outputs');

## Render backends and vector export

The series pass goes through a render backend port. `renderer: 'auto'` takes the WebGL2
backend from `openalgo-charts/webgl` where the device has WebGL2 and the 2D path
otherwise, decided once at construction; `chart.rendererKind` reports which one is
painting. Pan and zoom: the candles are batched into one GPU draw call per frame and
composited into the pane's own canvas, so screenshots, drawings, axes and text are
unchanged.

<RunnableExample height={320} code={`const chart = lib.createChart(el, { renderer: 'auto' });
const bars = lib.generateBars(1700000000, 600, 3600);
chart.addSeries('candlestick').setData(bars);
chart.timeScale.fitContent(240);

// Decided once, at construction: 'webgl2' where WebGL2 works, else 'canvas2d'.
el.style.position = 'relative';
const tag = document.createElement('div');
tag.style.cssText = 'position:absolute;right:64px;bottom:30px;z-index:5;font:12px ui-monospace,monospace;background:rgba(0,0,0,.55);color:#e6edf3;padding:4px 8px;border-radius:6px;pointer-events:none';
function show(note) { tag.textContent = 'rendererKind: ' + chart.rendererKind + (note ? '  ' + note : ''); }
show(lib.isWebGL2Supported() ? '' : '(no WebGL2 on this device)');
// A lost context moves the chart to the 2D path for the session and says so once.
chart.on('renderer:fallback', function (e) { show('(fell back: ' + e.reason + ')'); });
el.appendChild(tag);
return chart;`} />

`chart.exportSVG()` writes the same frame as a standalone vector document: axis labels
and tags stay text, lines stay lines, and nothing transient (crosshair, hover, drag) is in
it. The right half below is the string the left half returned, inlined as markup and
re-exported as you pan or zoom the live chart.

<RunnableExample height={360} code={`el.style.display = 'flex';
el.style.gap = '10px';
const live = document.createElement('div');
live.style.cssText = 'flex:1 1 0;min-width:0;height:100%;position:relative';
const out = document.createElement('div');
out.style.cssText = 'flex:1 1 0;min-width:0;height:100%;overflow:hidden;border:1px dashed #5b6478;border-radius:6px;line-height:0';
el.appendChild(live);
el.appendChild(out);

const chart = lib.createChart(live);
const bars = lib.generateBars(1700000000, 120, 3600);
chart.addSeries('candlestick').setData(bars);
chart.addPriceLine({ price: bars[bars.length - 1].close, color: '#f5a623', lineWidth: 1, dashed: true, id: 'last' });
chart.timeScale.fitContent(120);

// One export per frame at most: the string is the whole paint, so exporting
// on every zoom event during a wheel glide would repeat the work.
let queued = false;
function exportNow() {
  // Sized to the target panel rather than the live chart: the chart lays
  // itself out for the export and puts the live layout back before returning.
  // queued stays set until the export is done, so nothing the export itself
  // does can schedule the next one: an export that re-queued on its own
  // relayout would run once per frame for as long as the page is open.
  try { out.innerHTML = chart.exportSVG({ width: out.clientWidth, height: out.clientHeight }); }
  finally { queued = false; }
}
function queueExport() { if (!queued) { queued = true; requestAnimationFrame(exportNow); } }
chart.on('resize', queueExport);   // the first measured layout, and any later one
chart.on('zoom', queueExport);
chart.on('pan', queueExport);
requestAnimationFrame(function () { requestAnimationFrame(queueExport); });
return chart;`} />

The options, the fallback rules and the serialising context behind the export are on
[Render Backends & SVG Export](/docs/rendering).

<RunnableExample hideCode height={300} tiers={['draw']} code={`const chart = lib.createChart(el);

chart.addSeries('candlestick').setData(bars);

const last = bars[bars.length - 1].close;
// Risk-based sizing: qty = (accountSize x risk%) / stop distance.
new lib.DrawingController(chart).add({
  tool: 'long-position', paneIndex: 0,
  points: [
    { time: bars[70].time, price: last },
    // 2% up against 1% down: the same 2:1 the label reports, but both legs stay
    // inside the price scale's autoscale margin. At 4%/2% the target zone and
    // its label sat above the visible range and were clipped away.
    { time: bars[110].time, price: last * 1.02 },
    { time: bars[110].time, price: last * 0.99 },
  ],
  style: { color: '#8892a6', accountSize: 100000, risk: 1, showLabels: true },
});
return chart;`} />
